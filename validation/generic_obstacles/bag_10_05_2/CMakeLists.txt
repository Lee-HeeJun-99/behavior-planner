cmake_minimum_required(VERSION 3.12)
project(bag_obstacle_audit)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation search)
find_package(pcl_conversions REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(bag_obstacle_audit bag_obstacle_audit.cpp)
target_include_directories(bag_obstacle_audit PRIVATE "${REPO_ROOT}/src/lidar/include" ${PCL_INCLUDE_DIRS})
ament_target_dependencies(bag_obstacle_audit rclcpp std_msgs pcl_conversions geometry_msgs sensor_msgs visualization_msgs)
target_link_libraries(bag_obstacle_audit ${PCL_LIBRARIES} sqlite3)
